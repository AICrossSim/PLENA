//DO NOT USE FIX x0, x1, x2, x3, x4 , these are needed for offsets. So x5, x6, x7 could be used for intermediate steps
// just an idea with the pointers for loop indices
S_LD_FIX x1, 8, x5; 		//x5 = b
S_ADDI_FIX x5, x5, 1;				// x5+=1
S_ST_FIX x1, 8, x5;			// b = x5


S_ST_FIX x1, 8, x0;			    		    
; Loop batches i.e. FIX_MEM[ FIX [x1] ] , FIX_MEM[x1 + 8] is the index
	S_ST_FIX x1, 9, x0;	
	;Loop n_heads i.e. FIX_MEM[ FIX[x1] + 2],  FIX_MEM[x1 + 9] is the index
		S_ST_FIX x1, 10, x0;
		;Loop num_chunks i.e. FIX[ x1 + 6], FIX_MEM[x1 + 10] is the index

			;calculate address of A_cur , base + ((b * num_chunks + c) * block_len + l) * n_heads + h
			S_LD_FIX  x1, 8, x5; //load b into x5
			S_LD_FIX  x1, 6, x6; //load num_chunks into x6
			S_MUL_FIX x5, x5, x6; // x5 = x5 * x6 = b * num_chunks
			S_LD_FIX  x1, 10, x6; // load c into x6
			S_ADD_FIX x5, x5, x6; // x5 = b * num_chunks + c 
			S_LD_FIX x1, 5, x6;   // load block_len into x6
			S_MUL_FIX x5, x5, x6; // x5 = (b * num_chunks + c ) * block_len
			S_LD_FIX x1, 11, x6;  // load l into x6
			S_ADD_FIX x5, x5, x6; // x5 = (b * num_chunks + c ) * block_len + l
			S_LD_FIX x1, 2, x6;   // load n_heads into x6
			S_MUL_FIX x5, x5, x6; // x5 = ((b * num_chunks + c) * block_len + l) * n_heads
			S_LD_FIX  x4, 0, x6;  // load base into x6
			S_ADD_FIX x5, x5, x6; // x5 = base + ((b * num_chunks + c) * block_len + l) * n_heads
			S_LD_FIX  x1, 9, x6;  // load h into x6
			S_ADD_FIX x5, x5, x6; // x5 = base + ((b * num_chunks + c) * block_len + l) * n_heads + h , the correct address of A_cur

			;set stride size and fetch into VSRAM
			S_LD_FIX x1, 2, x6;   // load n_heads into x6
			C_SET_STRIDE_REG x6; //set stride size to n_heads
			S_LD_FIX x2, 2, x6; //x6 = A_cur VSRAM address
			H_PREFETCH_V_S x5, x6, 0; Vector SRAM [x6] = HBM[x5]

			;calculate address of B_cur , base + (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state
			;x5 already stores base + ((b * num_chunks + c) * block_len + l) * n_heads + h
			S_LD_FIX x4, 0, x6; //load A_cur base into x6
			S_SUB_FIX x5, x5, x6; // x5 = ((b * num_chunks + c) * block_len + l) * n_heads + h
			S_LD_FIX x1, 4, x6; // x6 = d_state
			S_MUL_FIX x5, x5, x6; // x5 = (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state
			S_LD_FIX x4, 1, x6; // x6 =  B_cur base
			S_ADD_FIX x5, x5, x6; // x5 = base + (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state , the correct address of B_cur

			;//find B_cur stride size & fetch into MSRAM
			S_LD_FIX x1, 2, x6; //x6 = n_heads
			S_LD_FIX x1, 4, x7; //x7 = d_states
			S_MUL_FIX x6, x6, x7; //x6 = n_heads * d_state
			C_SET_STRIDE_REG x6;  //stride size = n_heads * d_state
			S_LD_FIX x3, 0, x6; //x6 = x3 = B_cur MSRAM address
			H_PREFETCH_M_S x5, x6, 0; //Matrix SRAM [x6] = HBM[x5]

			;calculate address of C_cur , base + (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state
			S_LD_FIX x4, 1, x6; //x6 = B_cur base
			S_SUB_FIX x5, x5, x6; // x5 = (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state
			S_LD_FIX x4, 2, x6; //x6 = C_cur base
			S_ADD_FIX x5, x5, x6;// x5 = base + (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state , correct address for C_cur

			;C_cur has the same stride size as B_cur
			S_LD_FIX x3, 1, x6; //x6 = C_cur MSRAM address
			H_PREFETCH_M_S x5, x6, 0; //Matrix SRAM [x6] = HBM[x5] 

			;calculate address of X_cur 
			S_LD_FIX x4, 2, x6; // x6 = C_cur base
			S_SUB_FIX x5, x5, x6; // x5 = (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_state
			S_LD_FIX x1, 4, x6; // x6 = d_state
			S_DIV_FIX x5, x5, x6; //x5 = x5 / x6 = (((b * num_chunks + c) * block_len + l) * n_heads + h)
			S_LD_FIX x1, 3, x6; // x6 = d_head
			S_MUL_FIX x5, x5, x6; //x5 = (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_head
			S_LD_FIX x4, 3, x6; // x6 = X_cur base
			S_ADD_FIX x5, x5, x6;// x5 = base + (((b * num_chunks + c) * block_len + l) * n_heads + h) * d_head , correct address for X_cur

			;find X_cur stride size & fetch into MSRAM
			S_LD_FIX x1, 3, x6; //x6 = d_head
			S_LD_FIX x1, 2, x7; //x7 = n_heads
			S_MUL_FIX x6, x6, x7; //x6 = d_head * n_heads
			C_SET_STRIDE_REG x6;  //stride size = d_head * n_heads
			S_LD_FIX x3, 2, x6; // x6 = X_cur MSRAM address
			H_PREFETCH_M_S x5, x6, 0; //Matrix SRAM [x6] = HBM[x5]
